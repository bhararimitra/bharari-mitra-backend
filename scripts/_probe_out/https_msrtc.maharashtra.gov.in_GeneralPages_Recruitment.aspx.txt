<script>
    // PDF.js worker
    pdfjsLib.GlobalWorkerOptions.workerSrc =
      'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';

    // Render cancellation token
    let renderRequest = 0;

    async function loadPDF(url) {
        const myReq = ++renderRequest; // cancel previous renders
        const viewer = document.getElementById('pdfViewer');
        viewer.innerHTML = '<div class="p-3">Loading…</div>';

        try {
            const loadingTask = pdfjsLib.getDocument({ url: url }); // pass as object
            const pdf = await loadingTask.promise;

            if (myReq !== renderRequest) return; // aborted
            viewer.innerHTML = '';

            for (let pageNum = 1; pageNum <= pdf.numPages; pageNum++) {
                const page = await pdf.getPage(pageNum);
                if (myReq !== renderRequest) return; // aborted mid-way

                const viewport = page.getViewport({ scale: 1.5 });
                const canvas = document.createElement('canvas');
                canvas.width = viewport.width;
                canvas.height = viewport.height;

                await page.render({ canvasContext: canvas.getContext('2d'), viewport }).promise;
                if (myReq !== renderRequest) return; // aborted after render

                viewer.appendChild(canvas);
            }
        } catch (err) {
            console.error('PDF loading error:', err);
            viewer.innerHTML = '<p class="text-danger p-3">⚠️ PDF लोड करताना त्रुटी आली.</p>';
        }
    }

    // Initialize once (WebForms/UpdatePanel safe)
    document.addEventListener('DOMContentLoaded', function () {
        if (window._pdfModalInit) return;
        window._pdfModalInit = true;

        const modalEl = document.getElementById('pdfModal');
        const bsModal = new bootstrap.Modal(modalEl);

        // Delegate clicks for any element with [data-pdf]
        document.addEventListener('click', function (e) {
            const link = e.target.closest('[data-pdf]');
            if (!link) return;

            e.preventDefault(); // prevent navigation
            const url = link.getAttribute('data-pdf');
            if (!url) return;

            loadPDF(url);
            bsModal.show();
        });

        // Clear & cancel when modal hides
        modalEl.addEventListener('hidden.bs.modal', function () {
            renderRequest++; // cancels any in-flight load
            document.getElementById('pdfViewer').innerHTML = '';
        });
    });
</script>
